Rounds.jsx
import { useEffect, useState } from "react";

export default function Rounds({ topic, timeLimit, onTimeout }) {
  const [timeLeft, setTimeLeft] = useState(timeLimit);
  const [count, setCount] = useState(0);

  useEffect(() => {
    const timer = setInterval(() => {
      setTimeLeft((t) => {
        if (t <= 1) {
          clearInterval(timer);
          onTimeout(count);
          return 0;
        }
        return t - 1;
      });
    }, 1000);

    return () => clearInterval(timer);
  }, [count, onTimeout]);

  return (
    <>
      <h2>{topic}</h2>
      <h1>{count}</h1>
      <button onClick={() => setCount((c) => c + 1)}>+</button>
      <p>{timeLeft}s</p>
    </>
  );
}

Settings.jsx
import { useState } from "react";
import { categories, attributes } from "../data/categories";

export default function Settings({ onStart }) {
  const [mode, setMode] = useState("category");
  const [selected, setSelected] = useState([]);
  const [timer, setTimer] = useState(30);

  const options =
    mode === "category"
      ? categories
      : mode === "attribute"
      ? attributes
      : [...categories, ...attributes];

  function toggle(item) {
    setSelected((s) =>
      s.includes(item) ? s.filter((i) => i !== item) : [...s, item]
    );
  }

  return (
    <>
      <h2>Settings</h2>

      <div>
        <label>
          <input
            type="radio"
            checked={mode === "category"}
            onChange={() => setMode("category")}
          />
          Category
        </label>
        <label>
          <input
            type="radio"
            checked={mode === "attribute"}
            onChange={() => setMode("attribute")}
          />
          Attribute
        </label>
        <label>
          <input
            type="radio"
            checked={mode === "both"}
            onChange={() => setMode("both")}
          />
          Both
        </label>
      </div>

      <div style={{ marginTop: 16 }}>
        {options.map((o) => (
          <button
            key={o}
            onClick={() => toggle(o)}
            style={{
              margin: 4,
              background: selected.includes(o) ? "#6200ee" : "#ddd",
              color: selected.includes(o) ? "white" : "black",
            }}
          >
            {o}
          </button>
        ))}
      </div>

      <div style={{ marginTop: 16 }}>
        <label>Timer (seconds)</label>
        <input
          type="number"
          min="5"
          value={timer}
          onChange={(e) => setTimer(Number(e.target.value))}
          style={{ width: "100%", marginTop: 8 }}
        />
      </div>

      <button
        disabled={selected.length === 0}
        onClick={() => onStart({ selected, timer })}
        style={{ marginTop: 20 }}
      >
        Start
      </button>
    </>
  );
}

Timesout.jsx
export default function Timesout({ last, isLast, onNext, onFinish }) {
  if (!last) return null;

  return (
    <>
      <h2>Time’s up</h2>
      <p>{last.topic}</p>
      <p>Score: {last.score}</p>

      {!isLast ? (
        <button onClick={onNext}>Continue</button>
      ) : (
        <button onClick={onFinish}>Show Result</button>
      )}
    </>
  );
}

App.jsx
import { useState } from "react";

import Home from "./pages/Home";
import Settings from "./pages/Settings";
import Rounds from "./pages/Rounds";
import Timesout from "./pages/Timesout";
import Result from "./pages/Result";

export default function App() {
  const [page, setPage] = useState("home");
  const [topics, setTopics] = useState([]);
  const [current, setCurrent] = useState(0);
  const [results, setResults] = useState([]);
  const [timeLimit, setTimeLimit] = useState(30);

  function startGame({ selected, timer }) {
    setTopics(selected);
    setTimeLimit(timer);
    setResults([]);
    setCurrent(0);
    setPage("round");
  }

  function endRound(score) {
    setResults((r) => [...r, { topic: topics[current], score }]);
    setPage("timesout");
  }

  function next() {
    if (current + 1 < topics.length) {
      setCurrent((c) => c + 1);
      setPage("round");
    } else {
      setPage("result");
    }
  }

  function resetGame() {
    setPage("home");
    setTopics([]);
    setCurrent(0);
    setResults([]);
  }

  return (
    <div className="container">
      <div className="card">
        {page === "home" && <Home onStart={() => setPage("settings")} />}

        {page === "settings" && <Settings onStart={startGame} />}

        {page === "round" && (
          <Rounds
            topic={topics[current]}
            timeLimit={timeLimit}
            onTimeout={endRound}
          />
        )}

        {page === "timesout" && (
          <Timesout
            last={results[results.length - 1]}
            isLast={current === topics.length - 1}
            onNext={next}
            onFinish={() => setPage("result")}
          />
        )}

        {page === "result" && (
          <Result results={results} onHome={resetGame} />
        )}
      </div>
    </div>
  );
}
